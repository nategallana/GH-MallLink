        private void LoadData()
        {
            try
            {
                _context?.Dispose();
                _context = new AppDbContext();

                // Ensure SQLite Database & Table exist
                _context.Database.EnsureCreated();

                // Seed default values if table is empty
                if (!_context.PaymentMethods.Any())
                {
                    _context.PaymentMethods.AddRange(
                        new PaymentMethod { MethodName = "CASH", GhCode = "01", IsDefault = true, IsActive = true, CreatedAt = DateTime.Now },
                        new PaymentMethod { MethodName = "CARD", GhCode = "02", IsDefault = false, IsActive = true, CreatedAt = DateTime.Now }
                    );
                    _context.SaveChanges();
                }

                var methods = _context.PaymentMethods.OrderBy(p => p.Id).ToList();

                grid.DataSource = null;
                grid.AutoGenerateColumns = true;
                grid.DataSource = methods;

                // Column formatting after binding
                if (grid.Columns["Id"] != null)
                {
                    grid.Columns["Id"].ReadOnly = true;
                    grid.Columns["Id"].Width = 45;
                }
                if (grid.Columns["CreatedAt"] != null)
                {
                    grid.Columns["CreatedAt"].ReadOnly = true;
                    grid.Columns["CreatedAt"].Width = 140;
                }

                lblStatus.Text = $"{methods.Count} payment method(s) loaded.";
                lblStatus.ForeColor = Color.Gray;
            }
            catch (Exception ex)
            {
                lblStatus.Text = "Database loading failed.";
                lblStatus.ForeColor = Color.Red;
                MessageBox.Show($"Error loading PaymentMethods table:\n{ex.Message}", "Database Connection Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnAdd_Click(object sender, EventArgs e)
        {
            try
            {
                var newMethod = new PaymentMethod
                {
                    MethodName = "NEW_PAYMENT",
                    GhCode = "00",
                    IsDefault = false,
                    IsActive = true,
                    CreatedAt = DateTime.Now
                };

                _context.PaymentMethods.Add(newMethod);
                _context.SaveChanges();
                LoadData();
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Unable to insert record:\n{ex.Message}", "Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void BtnDelete_Click(object sender, EventArgs e)
        {
            if (grid.SelectedRows.Count == 0)
            {
                MessageBox.Show("Please select a row to delete.", "Selection Required", MessageBoxButtons.OK, MessageBoxIcon.Information);
                return;
            }

            var confirm = MessageBox.Show(
                $"Delete {grid.SelectedRows.Count} selected record(s)?",
                "Confirm Delete", MessageBoxButtons.YesNo, MessageBoxIcon.Warning);

            if (confirm != DialogResult.Yes) return;

            foreach (DataGridViewRow row in grid.SelectedRows)
            {
                if (row.DataBoundItem is PaymentMethod item)
                {
                    var entity = _context.PaymentMethods.Find(item.Id);
                    if (entity != null)
                    {
                        _context.PaymentMethods.Remove(entity);
                    }
                }
            }

            _context.SaveChanges();
            LoadData();
        }

        private void BtnSave_Click(object sender, EventArgs e)
        {
            try
            {
                grid.EndEdit();
                _context.SaveChanges();
                lblStatus.Text = "Changes saved successfully!";
                lblStatus.ForeColor = Color.Green;
            }
            catch (Exception ex)
            {
                MessageBox.Show($"Failed to save changes:\n{ex.Message}", "Save Error", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }
